async function generateStory() { const output = document.getElementById('storyOutput'); const charA = document.getElementById('charA').value; const charB = document.getElementById('charB').value; const relation = document.getElementById('relation').value; const intensity = document.getElementById('intensity').value; const apiKey = document.getElementById('apiKey').value; // 這裡貼上你的 hf_... Token if (!apiKey) { alert("兄弟，記得在左下角輸入 Hugging Face 的 Token 喔！"); return; } output.innerHTML = `正在連線至 Hugging Face 伺服器...`; try { // 使用 Mistral 模型，這是免費且強大的選擇 const response = await fetch("https://api-inference.huggingface.co/models/mistralai/Mistral-7B-Instruct-v0.3", { method: "POST", headers: { "Authorization": `Bearer ${apiKey}`, "Content-Type": "application/json" }, body: JSON.stringify({ "inputs": `[INST] 你是一位擅長情感張力與肢體描寫的小說家。 請撰寫一段關於 ${selectedType} 的故事。 角色：${charA} 與 ${charB}。關係：${relation}。 激烈程度：${intensity}%（請根據程度加強氛圍描述）。 請用繁體中文撰寫，注重角色的內心戲與感官細節。 [/INST]`, "parameters": {"max_new_tokens": 800, "temperature": 0.7} }) }); const result = await response.json(); // Hugging Face 的回傳格式有時候會包含原始輸入，我們只取生成的文字 let generatedText = result[0].generated_text; const splitText = generatedText.split("[/INST]"); output.innerHTML = splitText[splitText.length - 1].trim(); } catch (error) { output.innerHTML = "連線失敗！可能是模型正在讀取中（需等約30秒），或者 Token 錯誤。錯誤訊息：" + error.message; } }
